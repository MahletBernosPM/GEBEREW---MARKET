02

Role Enum

CREATE TYPE "Role" AS ENUM (
  'FARMER',       -- submits crop listings, queries prices
  'BUYER',        -- browses listings, contacts farmers
  'COOPERATIVE',  -- submits listings on behalf of multiple farmers
  'OPERATOR',     -- verifies prices, region-scoped access
  'ADMIN'         -- full access to all tables
);

-- Insert a market with GPS coordinates (longitude first, then latitude)

  ST_SetSRID(ST_MakePoint(38.7369, 9.0192), 4326)

-- Query coordinates back out
SELECT name, ST_X(location) AS lng, ST_Y(location) AS lat FROM markets;
